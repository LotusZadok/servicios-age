# Fase 1: Herramienta Interna Web — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the fase 1 internal web tool (Next.js + Supabase) that replaces the 2023 Android/Flask/SQL Server system: CRUD for clients/machines/brands/machine-types, admin/empleado roles enforced via RLS, a catalog CSV export, and a first deploy to Vercel.

**Architecture:** Single Next.js 15 (App Router, TypeScript) project. Supabase provides Postgres, Auth, and Row Level Security. Server Actions and Route Handlers replace the old Flask endpoints; Supabase's generated TypeScript types give end-to-end type safety. No ORM — the Supabase JS client is used directly.

**Tech Stack:** Next.js 15 (App Router), TypeScript, Tailwind CSS, @supabase/ssr, @supabase/supabase-js, Vitest (unit), Playwright (E2E), Supabase CLI + local Postgres via Docker for local dev.

**Prerequisites for whoever executes this plan:**
- Node.js 20+
- Docker Desktop running (required for local Supabase)
- A Supabase account (free tier) for the eventual cloud project used in Task 13

Spec reference: `docs/superpowers/specs/2026-07-23-gestion-maquinaria-web-fase1-design.md`

**Locked-in decisions made during planning (not re-litigated from the spec):**
- Report export format: **CSV** (opens in Excel), not PDF — simpler, no extra rendering dependency. Revisit if the business specifically needs a formatted PDF later.
- No ORM (no Drizzle) — the spec left this open; Supabase's generated types give sufficient type safety for this size of app.
- `profiles` table has no INSERT policy for regular authenticated users — profile rows are only ever created server-side via the service-role client (in the `createUser` action and the seed script), never through a client-exposed path.

-- profiles: extends auth.users with role info
create table public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name text not null,
  role text not null check (role in ('admin', 'empleado')),
  created_at timestamptz not null default now()
);

-- security-definer helper avoids recursive RLS lookups on profiles
create or replace function public.is_admin()
returns boolean
language sql
security definer
set search_path = public
stable
as $$
  select exists (
    select 1 from public.profiles
    where id = auth.uid() and role = 'admin'
  );
$$;

  on public.profiles for select
  to authenticated

- [ ] **Step 5: Create the middleware**

Create `src/middleware.ts`:

```typescript
import { createServerClient } from '@supabase/ssr'
import { NextResponse, type NextRequest } from 'next/server'

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({ request })

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll()
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value))
          response = NextResponse.next({ request })
          cookiesToSet.forEach(({ name, value, options }) =>
            response.cookies.set(name, value, options)
          )
        },
      },
    }
  )

  const {
    data: { user },
  } = await supabase.auth.getUser()

  const isAuthRoute = request.nextUrl.pathname.startsWith('/login')
  const isProtectedRoute = request.nextUrl.pathname.startsWith('/dashboard')

  if (!user && isProtectedRoute) {
    return NextResponse.redirect(new URL('/login', request.url))
  }

  if (user && isAuthRoute) {
    return NextResponse.redirect(new URL('/dashboard', request.url))
  }

  return response
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
}
```

import { redirect } from 'next/navigation'
import { createClient } from '@/lib/supabase/server'
import type { Database } from '@/lib/database.types'

type Profile = Database['public']['Tables']['profiles']['Row']

export async function requireUser() {
  const supabase = await createClient()
  const {
    data: { user },
  } = await supabase.auth.getUser()

  if (!user) {
    redirect('/login')
  }

  const { data: profile } = await supabase
    .from('profiles')
    .select('*')
    .eq('id', user.id)
    .single()

  if (!profile) {
    redirect('/login')
  }

  return { user, profile: profile as Profile }
}

export async function requireAdmin() {
  const { user, profile } = await requireUser()

  if (profile.role !== 'admin') {
    redirect('/dashboard')
  }

  return { user, profile }
}
```

'use server'

import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'
import { createClient } from '@/lib/supabase/server'
import { requireUser } from '@/lib/auth'
import { canDelete } from '@/lib/permissions'

export async function createBrand(formData: FormData) {
  const supabase = await createClient()
  const name = formData.get('name') as string

  const { error } = await supabase.from('brands').insert({ name })

  if (error) {
    redirect(`/dashboard/brands/new?error=${encodeURIComponent(error.message)}`)
  }

  revalidatePath('/dashboard/brands')
  redirect('/dashboard/brands')
}

export async function updateBrand(id: string, formData: FormData) {
  const supabase = await createClient()
  const name = formData.get('name') as string

  const { error } = await supabase.from('brands').update({ name }).eq('id', id)

  if (error) {
    redirect(`/dashboard/brands/${id}/edit?error=${encodeURIComponent(error.message)}`)
  }

  revalidatePath('/dashboard/brands')
  redirect('/dashboard/brands')
}

export async function deleteBrand(id: string) {
  const { profile } = await requireUser()

  if (!canDelete(profile.role)) {
    throw new Error('No tienes permiso para eliminar.')
  }

  const supabase = await createClient()
  await supabase.from('brands').delete().eq('id', id)
  revalidatePath('/dashboard/brands')
}
```

'use server'

import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'
import { createAdminClient } from '@/lib/supabase/admin'
import { requireAdmin } from '@/lib/auth'

export async function createUser(formData: FormData) {
  await requireAdmin()

  const email = formData.get('email') as string
  const password = formData.get('password') as string
  const fullName = formData.get('full_name') as string
  const role = formData.get('role') as string

  const admin = createAdminClient()

  const { data, error } = await admin.auth.admin.createUser({
    email,
    password,
    email_confirm: true,
  })

  if (error || !data.user) {
    redirect(
      `/dashboard/users/new?error=${encodeURIComponent(error?.message ?? 'Error desconocido')}`
    )
  }

  const { error: profileError } = await admin.from('profiles').insert({
    id: data.user.id,
    full_name: fullName,
    role,
  })

  if (profileError) {
    redirect(`/dashboard/users/new?error=${encodeURIComponent(profileError.message)}`)
  }

  revalidatePath('/dashboard/users')
  redirect('/dashboard/users')
}

export async function updateUserRole(id: string, formData: FormData) {
  await requireAdmin()

  const role = formData.get('role') as string
  const admin = createAdminClient()

  const { error } = await admin.from('profiles').update({ role }).eq('id', id)

  if (error) {
    redirect(`/dashboard/users?error=${encodeURIComponent(error.message)}`)
  }

  revalidatePath('/dashboard/users')
}

```typescript
import { NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { requireUser } from '@/lib/auth'

function toCsvRow(values: (string | null | undefined)[]) {
  return values.map((v) => `"${(v ?? '').replace(/"/g, '""')}"`).join(',')
}

export async function GET() {
  await requireUser()
  const supabase = await createClient()

  const [{ data: clients }, { data: machines }, { data: brands }, { data: types }] =
    await Promise.all([
      supabase.from('clients').select('*').order('name'),
      supabase.from('machines').select('*, brands(name), machine_types(name)').order('name'),
      supabase.from('brands').select('*').order('name'),
      supabase.from('machine_types').select('*').order('name'),
    ])

  const lines: string[] = []

  lines.push('# Clientes')
  lines.push(toCsvRow(['Nombre', 'Correo', 'Teléfono']))
  for (const c of clients ?? []) {
    lines.push(toCsvRow([c.name, c.email, c.phone]))
  }

  lines.push('')
  lines.push('# Máquinas')
  lines.push(toCsvRow(['Nombre', 'Serie', 'Marca', 'Tipo']))
  for (const m of machines ?? []) {
    lines.push(toCsvRow([m.name, m.serial_number, m.brands?.name, m.machine_types?.name]))
  }

  lines.push('')
  lines.push('# Marcas')
  lines.push(toCsvRow(['Nombre']))
  for (const b of brands ?? []) {
    lines.push(toCsvRow([b.name]))
  }

  lines.push('')
  lines.push('# Tipos de maquinaria')
  lines.push(toCsvRow(['Nombre']))
  for (const t of types ?? []) {
    lines.push(toCsvRow([t.name]))
  }

  const csv = lines.join('\n')

  return new NextResponse(csv, {
    headers: {
      'Content-Type': 'text/csv; charset=utf-8',
      'Content-Disposition': 'attachment; filename="catalogo-agservicios.csv"',
    },
  })
}
```
